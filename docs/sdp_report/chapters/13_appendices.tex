\section{Frontend: AuthContext.jsx}

The AuthContext provides the global authentication state throughout the React application.
It wraps the Supabase \texttt{onAuthStateChange} listener and exposes \texttt{user},
\texttt{session}, \texttt{storeProfile}, \texttt{loading}, and \texttt{signOut} to all
child components via the \texttt{useAuth()} hook.

\begin{lstlisting}[language=JavaScript, caption={AuthContext.jsx — Authentication Provider},
label={lst:authcontext}]
import { useState, useEffect, useContext, createContext } from 'react'
import { supabase } from '../lib/supabase'

const AuthContext = createContext(null)

export function AuthProvider({ children }) {
  const [user,         setUser]         = useState(null)
  const [session,      setSession]      = useState(null)
  const [storeProfile, setStoreProfile] = useState(null)
  const [loading,      setLoading]      = useState(true)

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session)
      setUser(session?.user ?? null)
      if (session?.user) fetchStoreProfile(session.user.id)
      else setLoading(false)
    })

    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      (_event, session) => {
        setSession(session)
        setUser(session?.user ?? null)
        if (session?.user) fetchStoreProfile(session.user.id)
        else { setStoreProfile(null); setLoading(false) }
      }
    )
    return () => subscription.unsubscribe()
  }, [])

  async function fetchStoreProfile(userId) {
    try {
      const { data, error } = await supabase
        .from('Store Profiles')
        .select('*')
        .eq('user_id', userId)
        .single()
      if (error && error.code !== 'PGRST116') throw error
      setStoreProfile(data || null)
    } catch { setStoreProfile(null) }
    finally  { setLoading(false) }
  }

  async function signOut() {
    await supabase.auth.signOut()
    setStoreProfile(null)
  }

  return (
    <AuthContext.Provider
      value={{ user, session, storeProfile, loading, signOut,
               refreshProfile: () => user && fetchStoreProfile(user.id) }}>
      {children}
    </AuthContext.Provider>
  )
}

export const useAuth = () => useContext(AuthContext)
\end{lstlisting}

\section{Frontend: lib/config.js — Webhook Trigger Utilities}

\begin{lstlisting}[language=JavaScript, caption={lib/config.js — n8n Webhook Helpers},
label={lst:config}]
const N8N_BASE = import.meta.env.VITE_N8N_BASE_URL || ''
const N8N_KEY  = import.meta.env.VITE_N8N_API_KEY  || ''

export async function triggerWebhook(path, method = 'GET') {
  const res = await fetch(`${N8N_BASE}/webhook/${path}`, {
    method,
    headers: { 'Content-Type': 'application/json' },
  })
  if (!res.ok) throw new Error(`Webhook error: ${res.status}`)
  return res.json()
}

export async function postWebhook(path, body) {
  const res = await fetch(`${N8N_BASE}/webhook/${path}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  })
  if (!res.ok) throw new Error(`Webhook error: ${res.status}`)
  return res.json()
}

export async function apiTriggerWorkflow(workflowId) {
  const res = await fetch(
    `${N8N_BASE}/api/v1/workflows/${workflowId}/activate`,
    { method: 'POST',
      headers: { 'X-N8N-API-KEY': N8N_KEY,
                 'Content-Type': 'application/json' } }
  )
  if (!res.ok) throw new Error(`API trigger error: ${res.status}`)
  return res.json()
}
\end{lstlisting}

\section{Frontend: Dashboard.jsx — Health Score and KPI Computation}

\begin{lstlisting}[language=JavaScript, caption={Dashboard.jsx — Core KPI and Health Score Logic},
label={lst:dashboard}]
function computeHealthScore(products) {
  const N  = products.length
  if (N === 0) return 100
  const nL = products.filter(p => p.risk_level === 'LOW').length
  const nM = products.filter(p => p.risk_level === 'MEDIUM').length
  const nH = products.filter(p => p.risk_level === 'HIGH').length
  const nO = products.filter(p => p.current_stock === 0).length
  const weighted = nL * 100 + nM * 70 + nH * 30 + nO * 5
  return Math.min(100, Math.max(0, Math.floor(weighted / N)))
}

// Inside Dashboard component:
const healthScore     = useMemo(() => computeHealthScore(products), [products])
const criticalProducts = useMemo(
  () => products
    .filter(p => ['HIGH','MEDIUM'].includes(p.risk_level))
    .sort((a, b) => (a.days_to_stockout ?? 999) - (b.days_to_stockout ?? 999))
    .slice(0, 5),
  [products]
)

fields (\texttt{\{\{products\_json\}\}} and \texttt{\{\{suppliers\_json\}\}}) are substituted
by n8n expressions at runtime.

\begin{lstlisting}[language={}  , caption={WF-05 Groq Llama 3.1 8B Zero-Shot Prompt Template},
label={lst:prompt}]
You are an inventory manager for a small Indian kirana store.
Your task is to generate reorder suggestions for at-risk products.

AT-RISK PRODUCTS:
{{products_json}}

AVAILABLE SUPPLIERS:
{{suppliers_json}}

Instructions:
- Generate exactly one suggestion per at-risk product.
- Select the best supplier based on grade and category match.
- Set suggested_qty = 30 * avg_daily_sales (round to nearest integer).
- Write a single-sentence reason in simple English.
- Return ONLY a valid JSON array. No markdown. No preamble.

Required output format:
[
  {
    "product_id": "string",

  }
